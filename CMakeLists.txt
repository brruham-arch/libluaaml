cmake_minimum_required(VERSION 3.18)
project(libluaaml C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT AML_INC OR NOT LUAJIT_DIR OR NOT GEN_DIR)
  message(FATAL_ERROR "set -DAML_INC -DLUAJIT_DIR -DGEN_DIR")
endif()
add_library(luaaml SHARED src/main.cpp)
target_include_directories(luaaml PRIVATE ${AML_INC} ${LUAJIT_DIR} ${GEN_DIR})
target_compile_options(luaaml PRIVATE -O2 -w -fvisibility=hidden)
target_link_libraries(luaaml PRIVATE ${LUAJIT_DIR}/libluajit.a log dl m)
target_link_options(luaaml PRIVATE -s -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,--gc-sections -Wl,-z,max-page-size=16384)
